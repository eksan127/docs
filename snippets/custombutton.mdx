export const CustomButton = ({ text, url, variant = 'primary' }) => {
  const baseStyles = "px-4 py-2 rounded-md font-medium transition-colors inline-block text-center";
  const variants = {
    primary: "bg-emerald-600 text-white hover:bg-emerald-700",
    secondary: "bg-gray-200 text-gray-800 hover:bg-gray-300",
    danger: "bg-red-600 text-white hover:bg-red-700"
  };

  return (
    <a href={url} className={`${baseStyles} ${variants[variant]}`}>
      {text}
    </a>
  );
};
